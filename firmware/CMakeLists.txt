cmake_minimum_required(VERSION 3.16)
project(pmars_host C)
enable_testing()
add_library(pmars_core pmars_core/src/pmars_core.c)
target_include_directories(pmars_core PUBLIC pmars_core/include)
target_compile_options(pmars_core PRIVATE -Wall -Wextra -Werror)
target_compile_options(pmars_core PRIVATE -fsanitize=address,undefined)
add_library(pmars_proto esp32/components/pmars_proto/proto.c)
target_include_directories(pmars_proto PUBLIC esp32/components/pmars_proto/include)
target_link_libraries(pmars_proto PUBLIC pmars_core)
target_compile_options(pmars_proto PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
add_executable(pmars_host_stdio host/pmars_host_stdio.c)
target_link_libraries(pmars_host_stdio PRIVATE pmars_proto pmars_core)
target_compile_options(pmars_host_stdio PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(pmars_host_stdio PRIVATE -fsanitize=address,undefined)
add_executable(test_core host_tests/test_core.c)
target_link_libraries(test_core PRIVATE pmars_core)
target_compile_options(test_core PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(test_core PRIVATE -fsanitize=address,undefined)
add_test(NAME pmars_host_core COMMAND test_core)
add_test(NAME pmars_host_stdio_smoke COMMAND pmars_host_stdio)
